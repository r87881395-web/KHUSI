];

let cart=JSON.parse(localStorage.getItem("khusiCart")||"[]"), active="All";

function filterCat(c){active=c;render()}

function render(){

 let q=document.getElementById("search").value.toLowerCase();

 let list=products.filter(p=>(active==="All"||p.cat===active)&&p.name.toLowerCase().includes(q));

 document.getElementById("result").textContent=list.length+" items";

 document.getElementById("products").innerHTML=list.length?list.map(p=>`<div class="card"><div class="pic">${p.icon}</div><div class="name">${p.name}</div><div><span class="price">₹${p.price}</span><span class="old">₹${p.old}</span></div><button class="add" onclick="add(${p.id})">ADD TO CART</button></div>`).join(""):`<div class="empty">No products found.</div>`;

 updateCount();

}

function add(id){let x=cart.find(i=>i.id===id);x?x.qty++:cart.push({id,qty:1});save();openCart()}

function save(){localStorage.setItem("khusiCart",JSON.stringify(cart));updateCount()}

function updateCount(){document.getElementById("count").textContent=cart.reduce((s,i)=>s+i.qty,0)}

function openCart(){document.getElementById("modal").classList.add("show");renderCart()}

function closeCart(){document.getElementById("modal").classList.remove("show")}

function renderCart(){

 let box=document.getElementById("cart"),total=0;

 if(!cart.length){box.innerHTML='<div class="empty">Your cart is empty.</div>';document.getElementById("total").textContent=0;return}

 box.innerHTML=cart.map(i=>{let p=products.find(x=>x.id===i.id);total+=p.price*i.qty;return `<div class="cart-row"><div class="cart-icon">${p.icon}</div><div style="flex:1"><b>${p.name}</b><div>₹${p.price} × ${i.qty}</div><div class="qty"><button onclick="change(${p.id},-1)">−</button> ${i.qty} <button onclick="change(${p.id},1)">+</button> <button onclick="removeItem(${p.id})">Remove</button></div></div></div>`}).join("");

 document.getElementById("total").textContent=total;

}

function change(id,n){let x=cart.find(i=>i.id===id);x.qty+=n;if(x.qty<=0)cart=cart.filter(i=>i.id!==id);save();renderCart();render()}

function removeItem(id){cart=cart.filter(i=>i.id!==id);save();renderCart();render()}

render();
